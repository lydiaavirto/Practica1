#include <Arduino.h>
#include "BBTimer.h"

const int pinADC = A0;

// Seleccionamos el temporizador por hardware TIMER0
BBTimer miTimer(BB_TIMER0);

// Bandera para la interrupción
volatile bool flagTimer = false;

// Función callback que llamará la interrupción hardware
void callbackTimer() {
  flagTimer = true;
}

void setup() {
  Serial.begin(9600);

  // Configura el timer para saltar cada 10.000.000 us (10 segundos)
  miTimer.setupTimer(10000000, callbackTimer);
  
  // Inicia el temporizador de hardware
  miTimer.timerStart();
}

void loop() {
  // Cuando transcurren los 10 segundos, se activa la bandera
  if (flagTimer) {
    flagTimer = false; // Reiniciamos la bandera

    // 1. Lectura del valor del ADC
    int valorADC = analogRead(pinADC);

    // 2. Cálculo de tensión en milivoltios
    long tension_mV = ((long)valorADC * 3300) / 1023;

    // 3. Formateo con sprintf()
    char buffer[64];
    sprintf(buffer, "[Timer 10s] ADC: %4d | Tension: %ld mV", valorADC, tension_mV);

    // 4. Impresión por consola
    Serial.println(buffer);
  }
}
